import{createTemplate,trackTemplate}from'./tracker.mjs';import{writeFile}from'node:fs/promises';import{createHash}from'node:crypto';
const W=128,H=128,pivot={x:64,y:64},radius=20,sigma=2,peak=220,n=512,origin={x:64,y:84};
function render(q,steps){const sum=new Float64Array(W*H);let sx=0,sy=0,angle=0;for(let k=0;k<steps;k++){const t=(k+.5)/steps,a=q(t),cx=64+radius*Math.sin(a),cy=64+radius*Math.cos(a);sx+=cx/steps;sy+=cy/steps;angle+=a/steps;for(let y=0;y<H;y++)for(let x=0;x<W;x++)sum[y*W+x]+=peak*Math.exp(-((x-cx)**2+(y-cy)**2)/(2*sigma*sigma))/steps;}const data=new Uint8ClampedArray(W*H*4);let mass=0,mx=0,my=0;for(let i=0;i<sum.length;i++){data[4*i]=data[4*i+1]=data[4*i+2]=sum[i];data[4*i+3]=255;mass+=data[4*i];mx+=(i%W)*data[4*i];my+=Math.floor(i/W)*data[4*i];}return{frame:{width:W,height:H,data},ideal:{x:sx,y:sy},scalarMean:angle,rasterCentroid:{x:mx/mass,y:my/mass},sha256:createHash('sha256').update(data).digest('hex')};}
const source=render(()=>0,1),template=createTemplate(source.frame,origin);if(template.status!=='ready')throw Error(template.status);const rows=[];
for(const [name,q] of [['stationary',()=>0],['quadratic0.2',t=>.2*t*t],['quadratic0.8',t=>.8*t*t],['quadratic2.4',t=>2.4*t*t],['complete-turn',t=>2*Math.PI*t]]){const exposure=render(q,n),r=trackTemplate(exposure.frame,template.template,origin,{searchRadius:48});rows.push({name,exposureSHA256:exposure.sha256,scalarMean:exposure.scalarMean,idealCentroid:exposure.ideal,rasterCentroid:exposure.rasterCentroid,idealAngle:Math.atan2(exposure.ideal.x-64,exposure.ideal.y-64),actual:r,acceptedAngle:r.point?Math.atan2(r.point.x-64,r.point.y-64):null});}
await writeFile(new URL('results.json',import.meta.url),JSON.stringify({scope:'Exploratory authored RGBA pixel arrays; unchanged maintained tracker; no camera or decoding claim',pixels:{width:W,height:H,sigma,peak,background:0,quantization:'Uint8ClampedArray after temporal average',pixelSample:'integer x,y, no spatial pixel-area integration'},exposure:{start:0,end:1,temporalMidpoints:n,brightness:'constant',cadence:'none; independently authored target exposures, no video cadence or shutter inferred'},pivot,radius,sourceHash:source.sha256,rows},null,2)+'\n');console.log(rows.map(x=>({name:x.name,status:x.actual.status,reason:x.actual.reason,score:x.actual.score,idealAngle:x.idealAngle,acceptedAngle:x.acceptedAngle,point:x.actual.point,candidate:x.actual.candidatePoint})));
